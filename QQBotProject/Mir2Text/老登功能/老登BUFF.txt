[@刷新BUFF]
{
#if
#act
;基础暴率
MOV N$真实战速 0
MOV N$真实体魄 1
MOV N$真实宝宝 0
MOV N$真实爆率 100
MOV N$真实经验 100
MOV S$爆率文本 基础爆率：100%\
MOV S$经验文本 基础经验：100%\

;老登活动
MUL N$真实爆率 2
INC S$爆率文本 老登活动爆率*2\

;等级叠加
DIV N$等级叠加 <$LEVEL> 10
INC N$真实战速 <$STR(N$等级叠加)>
DIV N$等级叠加 <$LEVEL> 5
INC N$真实体魄 <$STR(N$等级叠加)>
INC N$真实宝宝 <$LEVEL>
MOV S$战士BUFF文本 攻速+<$STR(N$真实战速)>(等级/10)\
MOV S$法师BUFF文本 防御+<$STR(N$真实体魄)>(等级/5)\
MOV S$道士BUFF文本 宝宝属性+<$STR(<$LEVEL>)>%(等级)\

;CDK叠加
;卡数X50
MUL N$CDK叠加 *CDK数量变量* 50
INC N$真实爆率 <$STR(N$CDK叠加)>
INC N$真实经验 <$STR(N$CDK叠加)>
INC S$爆率文本 群CDK爆率+<$STR(N$CDK叠加)>%（CDK数量：<$STR(*CDK数量变量*)>*50%）\
INC S$经验文本 群CDK经验+<$STR(N$CDK叠加)>%（CDK数量：<$STR(*CDK数量变量*)>*50%）\

;卡数除2
DIV N$CDK叠加 *CDK数量变量* 2
INC N$真实战速 <$STR(N$CDK叠加)>
INC S$战士BUFF文本 攻速+<$STR(N$CDK叠加)>(CDK数量/2)\

;原始卡数
INC N$真实体魄 <$STR(*CDK数量变量*)>
INC S$法师BUFF文本 防御+<$STR(*CDK数量变量*)>(CDK数量)\

;卡数X10
MUL N$CDK叠加 *CDK数量变量* 10
INC N$真实宝宝 <$STR(N$CDK叠加)>
INC S$道士BUFF文本 宝宝属性+<$STR(N$CDK叠加)>%(CDK数量*10)\

;实现部分
#IF
CheckJob Warrior
#ACT
CHANGESPEED 2 <$STR(N$真实战速)>

#IF
CheckJob WIZARD
#ACT
ChangeHumAbility 1 = <$STR(N$真实体魄)>
ChangeHumAbility 2 = <$STR(N$真实体魄)>
ChangeHumAbility 3 = <$STR(N$真实体魄)>
ChangeHumAbility 4 = <$STR(N$真实体魄)>

#IF
CheckJob TAOIST
#ACT
ChangeSlaveAbilityEX 0 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 1 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 2 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 3 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 4 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 5 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 6 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 7 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 8 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 9 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 10 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 11 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 12 +<$STR(N$真实宝宝)> 1
ChangeSlaveAbilityEX 13 +<$STR(N$真实宝宝)> 1
RecalcSlaveAbility 

#IF
#ACT
KILLMONBURSTRATE <$STR(N$真实爆率)> 0 1 1
KILLMONEXPRATE <$STR(N$真实经验)> 0 1 1

[@计算过程]
#IF
CheckJob Warrior
#SAY
战士属性加成:\
<$STR(S$战士BUFF文本)>\
总计:<$STR(N$真实战速)>\ \

#IF
CheckJob WIZARD
#SAY
法师属性加成:\
<$STR(S$法师BUFF文本)>\
总计:<$STR(N$真实体魄)>\ \

#IF
CheckJob TAOIST
#SAY
道士属性加成:\
<$STR(S$道士BUFF文本)>\
总计:<$STR(N$真实宝宝)>\ \

#IF
#SAY
爆率计算过程:\
<$STR(S$爆率文本)>\
总计:<$KILLMONBURSTRATE>\ \
经验计算过程:\
<$STR(S$经验文本)>\
总计:<$KILLMONEXPRATE>\ \

<返回/@老登帮助>
}
